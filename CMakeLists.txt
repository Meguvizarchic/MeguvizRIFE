cmake_minimum_required(VERSION 3.31)
project(MeguvizRIFE VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(MGR_ENABLE_JOB_OBSERVATION "Enable optional result-owned execution observations" OFF)
option(MGR_ENABLE_SHARED_PLAN "Build the private shared GPU plan used by the Android composition" OFF)
option(MGR_ENABLE_GROUPED_STATISTICS "Enable workgroup-local guard statistics for built-in Vulkan shaders" OFF)
set(MGR_SHARED_PLAN_ENABLED OFF)
if(MGR_ENABLE_SHARED_PLAN)
  set(MGR_SHARED_PLAN_ENABLED ON)
endif()
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(NOT TARGET MeguvizCompute::backend_ncnn_sdk)
  find_package(MeguvizCompute 0.1 CONFIG REQUIRED)
endif()
get_target_property(MGR_PRIVATE_FP MeguvizCompute::backend_ncnn_sdk INTERFACE_MGC_PRIVATE_SDK_FINGERPRINT)
if(NOT MGR_PRIVATE_FP MATCHES "^[0-9a-f]+$" OR NOT MGR_PRIVATE_FP)
  message(FATAL_ERROR "The Compute SDK must expose its exact private fingerprint")
endif()
file(GLOB_RECURSE identity_sources CONFIGURE_DEPENDS include/*.h src/*.cpp src/*.h src/*.hpp src/*.inc src/*.comp)
list(APPEND identity_sources "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" "${CMAKE_CURRENT_SOURCE_DIR}/cmake/build_identity.hpp.in")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${identity_sources})
list(SORT identity_sources)
set(identity_text "sdk=${MGR_PRIVATE_FP};compiler=${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION};system=${CMAKE_SYSTEM_NAME};abi=${ANDROID_ABI};config=${CMAKE_BUILD_TYPE};flags=${CMAKE_CXX_FLAGS};release=${CMAKE_CXX_FLAGS_RELEASE};stl=${CMAKE_MSVC_RUNTIME_LIBRARY}-${ANDROID_STL};job_observation=${MGR_ENABLE_JOB_OBSERVATION};shared_plan=${MGR_SHARED_PLAN_ENABLED};grouped_statistics=${MGR_ENABLE_GROUPED_STATISTICS}")
foreach(file IN LISTS identity_sources)
  file(SHA256 "${file}" content_hash)
  file(RELATIVE_PATH logical_path "${CMAKE_CURRENT_SOURCE_DIR}" "${file}")
  string(APPEND identity_text ";${logical_path}=${content_hash}")
endforeach()
string(SHA256 MGR_BUILD_FP "${identity_text}")
configure_file(cmake/build_identity.hpp.in generated/build_identity.hpp @ONLY)
configure_file(cmake/build-info.json.in build-info.json @ONLY)
set(legacy src/backends/ncnn/legacy)
set(shader_headers)
set(legacy_shaders upstream/warp upstream/warp_pack4 frame_preproc frame_postproc frame_fused frame_fused_rgb8 frame_reduce
  frame_gpu_unpack frame_gpu_image_read frame_gpu_statistics frame_gpu_patch frame_gpu_image_write frame_add_leaky)
if(MGR_ENABLE_GROUPED_STATISTICS)
  list(APPEND legacy_shaders frame_gpu_statistics_grouped)
endif()
foreach(shader IN LISTS legacy_shaders)
  get_filename_component(name "${shader}" NAME)
  set(header "${CMAKE_CURRENT_BINARY_DIR}/generated/${name}.comp.hex.h")
  add_custom_command(OUTPUT "${header}" COMMAND "${CMAKE_COMMAND}"
    "-DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}/${legacy}/${shader}.comp" "-DOUTPUT=${header}"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/${legacy}/embed_shader.cmake"
    DEPENDS "${legacy}/${shader}.comp" "${legacy}/embed_shader.cmake" VERBATIM)
  list(APPEND shader_headers "${header}")
endforeach()
add_library(meguviz_rife STATIC src/core/rife.cpp "${legacy}/frame_engine.cpp" "${legacy}/add_leaky.cpp" "${legacy}/upstream/rife_warp.cpp" ${shader_headers})
if(MGR_ENABLE_GROUPED_STATISTICS)
  target_compile_definitions(meguviz_rife PRIVATE MGR_ENABLE_GROUPED_STATISTICS=1)
endif()
if(MGR_ENABLE_JOB_OBSERVATION)
  target_compile_definitions(meguviz_rife PRIVATE MGR_ENABLE_JOB_OBSERVATION=1)
endif()
add_library(MeguvizRIFE::core ALIAS meguviz_rife)
set_target_properties(meguviz_rife PROPERTIES EXPORT_NAME core CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
set_target_properties(meguviz_rife PROPERTIES INTERFACE_MGC_PRIVATE_SDK_FINGERPRINT "${MGR_PRIVATE_FP}")
set_property(TARGET meguviz_rife APPEND PROPERTY COMPATIBLE_INTERFACE_STRING MGC_PRIVATE_SDK_FINGERPRINT)
target_include_directories(meguviz_rife PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
  PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(meguviz_rife PRIVATE MeguvizCompute::backend_ncnn_sdk PUBLIC MeguvizCompute::core)
if(MSVC)
  target_compile_options(meguviz_rife PRIVATE /EHsc /W4)
else()
  set_source_files_properties(src/core/rife.cpp "${legacy}/frame_engine.cpp" PROPERTIES COMPILE_OPTIONS -fexceptions)
  target_compile_options(meguviz_rife PRIVATE -Wall -Wextra)
endif()
add_library(meguviz_rife_ncnn INTERFACE)
add_library(MeguvizRIFE::ncnn ALIAS meguviz_rife_ncnn)
set_target_properties(meguviz_rife_ncnn PROPERTIES EXPORT_NAME ncnn)
target_link_libraries(meguviz_rife_ncnn INTERFACE meguviz_rife)
option(MGR_BUILD_CLI "Build the independent two-image reference CLI" ON)
set(MGR_REQUIRED_PRIVATE_SDK_FINGERPRINT "" CACHE STRING "Required Compute private SDK fingerprint")
if(MGR_REQUIRED_PRIVATE_SDK_FINGERPRINT)
  get_target_property(sdk_identity MeguvizCompute::backend_ncnn_sdk INTERFACE_MGC_PRIVATE_SDK_FINGERPRINT)
  if(NOT sdk_identity STREQUAL MGR_REQUIRED_PRIVATE_SDK_FINGERPRINT)
    message(FATAL_ERROR "RIFE private SDK fingerprint mismatch")
  endif()
endif()
if(MGR_BUILD_CLI)
  add_executable(mgr-cli src/cli/main.cpp)
  target_include_directories(mgr-cli PRIVATE src)
  target_include_directories(mgr-cli PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_link_libraries(mgr-cli PRIVATE meguviz_rife MeguvizCompute::backend_ncnn_sdk)
  if(MSVC)
    target_compile_options(mgr-cli PRIVATE /EHsc)
  else()
    # ncnn's interface flags must not disable exceptions in the CLI parser.
    set_source_files_properties(src/cli/main.cpp PROPERTIES COMPILE_OPTIONS -fexceptions)
  endif()
  install(TARGETS mgr-cli RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
install(TARGETS meguviz_rife meguviz_rife_ncnn EXPORT MeguvizRIFETargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES models/catalog/rife46.manifest.json models/catalog/profiles.json models/catalog/shader-contract.json LICENSE README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/MeguvizRIFE)
install(FILES tools/prepare_shaderpack.py DESTINATION ${CMAKE_INSTALL_DATADIR}/MeguvizRIFE/tools)
install(FILES docs/adr/0002-compute-gpu-guards.md DESTINATION ${CMAKE_INSTALL_DATADIR}/MeguvizRIFE/design)
install(FILES docs/adr/0003-job-observation.md DESTINATION ${CMAKE_INSTALL_DATADIR}/MeguvizRIFE/design)
install(DIRECTORY examples/external_consumer/ DESTINATION ${CMAKE_INSTALL_DATADIR}/MeguvizRIFE/examples/external_consumer)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/build-info.json" DESTINATION ${CMAKE_INSTALL_DATADIR}/MeguvizRIFE)
install(FILES "${legacy}/NOTICE.md" "${legacy}/upstream/LICENSE" DESTINATION ${CMAKE_INSTALL_DATADIR}/MeguvizRIFE/notices)
configure_package_config_file(cmake/MeguvizRIFEConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/MeguvizRIFEConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MeguvizRIFE)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/MeguvizRIFEConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT MeguvizRIFETargets NAMESPACE MeguvizRIFE:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MeguvizRIFE)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/MeguvizRIFEConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/MeguvizRIFEConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MeguvizRIFE)
option(MGR_BUILD_TESTS "Build independent C ABI boundary checks" ON)
if(MGR_BUILD_TESTS)
  enable_testing()
  add_executable(mgr_c_consumer tests/c_consumer.c)
  target_link_libraries(mgr_c_consumer PRIVATE MeguvizRIFE::core)
  set_target_properties(mgr_c_consumer PROPERTIES LINKER_LANGUAGE CXX)
  add_test(NAME public_c_model_bounds COMMAND mgr_c_consumer)
  add_executable(mgr_observation_abi tests/observation_abi.c)
  target_link_libraries(mgr_observation_abi PRIVATE MeguvizRIFE::core)
  set_target_properties(mgr_observation_abi PROPERTIES LINKER_LANGUAGE CXX)
  add_test(NAME public_observation_abi_boundaries COMMAND mgr_observation_abi)
  # Explicit opt-in real-model exercise; not a CTest because model bytes are
  # supplied by the caller and are never bundled in the SDK.
  add_executable(mgr_observation_model tests/observation_model.cpp)
  target_link_libraries(mgr_observation_model PRIVATE MeguvizRIFE::core)
  if(MGR_ENABLE_JOB_OBSERVATION)
    target_compile_definitions(mgr_observation_model PRIVATE MGR_ENABLE_JOB_OBSERVATION=1)
  endif()
  if(MSVC)
    target_compile_options(mgr_observation_model PRIVATE /EHsc)
  endif()
  # These regressions use caller-provided research model bytes, never bundled.
  add_executable(mgr_context_lifecycle tests/context_lifecycle.cpp)
  target_link_libraries(mgr_context_lifecycle PRIVATE MeguvizRIFE::core MeguvizCompute::backend_ncnn_sdk)
  add_executable(mgr_callback_pool_reentry tests/callback_pool_reentry.cpp)
  target_link_libraries(mgr_callback_pool_reentry PRIVATE MeguvizRIFE::core MeguvizCompute::backend_ncnn_sdk)
  add_executable(mgr_alpha_boundary tests/alpha_boundary.cpp)
  target_link_libraries(mgr_alpha_boundary PRIVATE MeguvizRIFE::core)
  if(MGR_ENABLE_JOB_OBSERVATION)
    target_compile_definitions(mgr_alpha_boundary PRIVATE MGR_ENABLE_JOB_OBSERVATION=1)
  endif()
  if(MSVC)
    target_compile_options(mgr_context_lifecycle PRIVATE /EHsc)
    target_compile_options(mgr_callback_pool_reentry PRIVATE /EHsc)
    target_compile_options(mgr_alpha_boundary PRIVATE /EHsc)
  endif()
  set(MGR_TEST_MODEL_DIR "" CACHE PATH "Optional caller-supplied rife-v4.6 model directory for lifecycle tests")
  if(MGR_TEST_MODEL_DIR)
    if(NOT EXISTS "${MGR_TEST_MODEL_DIR}/flownet.param" OR NOT EXISTS "${MGR_TEST_MODEL_DIR}/flownet.bin")
      message(FATAL_ERROR "MGR_TEST_MODEL_DIR must contain flownet.param and flownet.bin")
    endif()
    set(mgr_test_manifest "${CMAKE_CURRENT_SOURCE_DIR}/models/catalog/rife46.manifest.json")
    add_test(NAME context_lifecycle_with_model COMMAND mgr_context_lifecycle "${MGR_TEST_MODEL_DIR}" "${mgr_test_manifest}")
    add_test(NAME callback_pool_reentry_with_model COMMAND mgr_callback_pool_reentry "${MGR_TEST_MODEL_DIR}" "${mgr_test_manifest}")
    add_test(NAME alpha_boundary_with_model COMMAND mgr_alpha_boundary "${MGR_TEST_MODEL_DIR}" "${mgr_test_manifest}")
    if(NCNN_VULKAN)
      add_executable(mgr_upload_lifecycle tests/upload_lifecycle.cpp)
      target_link_libraries(mgr_upload_lifecycle PRIVATE MeguvizRIFE::core MeguvizCompute::backend_ncnn_sdk)
      if(MSVC)
        target_compile_options(mgr_upload_lifecycle PRIVATE /EHsc)
      endif()
      add_test(NAME upload_lifecycle_with_model COMMAND mgr_upload_lifecycle "${MGR_TEST_MODEL_DIR}" "${mgr_test_manifest}")
      set_tests_properties(upload_lifecycle_with_model PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 120)
    endif()
  endif()
  add_executable(mgr-legacy-oracle tests/oracle/main.cpp tests/oracle/frame_engine.cpp tests/oracle/upstream/rife_warp.cpp ${shader_headers})
  target_include_directories(mgr-legacy-oracle PRIVATE tests/oracle "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_link_libraries(mgr-legacy-oracle PRIVATE MeguvizCompute::backend_ncnn_sdk)
  if(MSVC)
    target_compile_options(mgr-legacy-oracle PRIVATE /EHsc)
  else()
    set_source_files_properties(tests/oracle/frame_engine.cpp PROPERTIES COMPILE_OPTIONS -fexceptions)
  endif()
endif()
